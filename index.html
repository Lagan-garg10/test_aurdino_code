Wiring

  • Thermistor Wire 1: Connect to Arduino 5V.
  • Thermistor Wire 2: Connect to Arduino A0 and one leg of the 10kΩ resistor.
  • 10kΩ Resistor (other leg): Connect to Arduino GND.
  • Buzzer Positive (+): Connect to Arduino Digital Pin 8.
  • Buzzer Negative (-): Connect to Arduino GND.

Arduino Code


#include <math.h>

// Pin Definitions
const int thermistorPin = A0;
const int buzzerPin = 8;

// Temperature Threshold in Celsius
const float tempThreshold = 30.0;

// Thermistor Constants for 10K NTC (Beta 3950)
const float seriesResistor = 10000.0;
const float nominalResistance = 10000.0;
const float nominalTemp = 25.0;
const float bCoefficient = 3950.0;

void setup() {
    Serial.begin(9600);
    pinMode(buzzerPin, OUTPUT);
    digitalWrite(buzzerPin, LOW);
}

void loop() {
    int adcValue = analogRead(thermistorPin);

    // Avoid division by zero
    if (adcValue <= 0 || adcValue >= 1023) {
        Serial.println("Invalid thermistor reading");
        digitalWrite(buzzerPin, LOW);
        delay(500);
        return;
    }

    // Convert ADC reading to resistance
    float resistance = seriesResistor /
        ((1023.0 / adcValue) - 1.0);

    // Calculate temperature
    float steinhart = resistance / nominalResistance;
    steinhart = log(steinhart);
    steinhart /= bCoefficient;
    steinhart += 1.0 / (nominalTemp + 273.15);
    steinhart = 1.0 / steinhart;

    float temperatureC = steinhart - 273.15;

    // Print temperature
    Serial.print("Temperature: ");
    Serial.print(temperatureC);
    Serial.println(" °C");

    // Buzzer Control
    if (temperatureC > tempThreshold) {
        digitalWrite(buzzerPin, HIGH);
    } else {
        digitalWrite(buzzerPin, LOW);
    }

    delay(500);
}
